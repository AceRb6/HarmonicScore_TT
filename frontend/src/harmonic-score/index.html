def transcribe_audio(file_path):
    # Inicializando modelo acústico
    model = HarmonicAI.load_model('polyphonic_v2')
    
    # Extracción de características (CQT)
    features = extract_spectrogram(file_path)
    
    # Predicción de notas y tiempos
    notes = model.predict(features)
    
    return generate_sheet_music(notes)
